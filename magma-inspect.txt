          damageEnemy(target, dmg*sp.meleeMult*(meleeCrit?1.8:1));
          fx('knifeSlash', pt);
          if(meleeCrit) fx('critFlash', pt);
        }
        break;
      }
      case 'asclepiusSyringe': {
        spawnPlayerShot(idx, target, 0.08, 0.26, 'syringe', ()=>{
          if(!target.alive) return;
          damageEnemy(target, dmg);
          if(Math.random() < sp.confuseChance){
            target.confusion = {t: sp.confuseDur, dirT:0, dir:1};
            fx('syringeHit', enemyRenderPos(target));
          } else {
            fx('streak', enemyRenderPos(target));
          }
        });
        break;
      }
      case 'crossingGate': {
        if(!state.gate && state.gateCooldownT<=0){
          spawnCrossingGate(target, sp);
        }
        break;
      }
      case 'devilShadow': {
        const cls = slottedClassOf(die.key);
        const T = devilShadowTierStats(cls);
        const rangeBand = T.includeClose ? 'map' : 'midfar';
        // Deliberately NOT using filterByRange here — its "never leave a die with nothing to
        // swing at" fallback would hand this die a close-range target before C4, which then
        // plays a full whiffed cast (tickDevilSwarm correctly refuses to damage them, since
        // they're out of the allowed band) — better to just skip the cast entirely that cycle.
        const rawList = aliveEnemies().filter(en=>inRangeBand(en, rangeBand));
        const chosenTarget = rawList.length ? pickTarget('random', rawList) : null;
        if(!chosenTarget) break;
        const origin = dieTileLaunchPos(idx);
        const tp3 = enemyRenderPos(chosenTarget);
        const coneAngle = Math.atan2(tp3.y-origin.y, tp3.x-origin.x);
        const wasClose = inRangeBand(chosenTarget, 'close');
        state.devilCasts.push({
          target: chosenTarget, origin, phase:'manifest', t:0,
          manifestDur:0.3, windupDur:0.35, releaseDur:1.5, neutralDur:0.8, fadeDur:0.3,
          clawResolveT:0.30, // moment within the 'neutral' phase the claw actually connects, if it fires
          cls, T, sp, dmg, aff:dieAff, // dmg already includes crit — never re-apply a crit multiplier below
          coneAngle, coneHalfAngle: sp.coneHalfAngle + T.coneWidthBonus,
          hitSet:new Set(), insideMap:new Map(),
          comboEligible: wasClose && T.comboEnabled, comboFired:false, resolved:false,
          bats:[], batSpawnTimer:0,
        });
        break;
      }
      default: damageEnemy(target, dmg); fx('streak', pt);
    }
    if(isCrit && sp && sp.kind!=='magnetPull') fx('critFlash', pt);
  }

  /* ============================ DEVIL SHADOW (bat swarm + close-range claw combo) ============================ */
  function devilShadowTierStats(cls){
    return {
      coneWidthBonus: cls>=3 ? 0.08 : 0,
      includeClose: cls>=4,
      stunDur: cls>=5 ? 1.85 : 1.5,
      comboEnabled: cls>=6,

  const DEVIL_BAT_AVG_SPEED = 200; // matches the average of the bat speed range used when spawning them
  function tickDevilSwarm(c, dt){
    const band = c.T.includeClose ? 'map' : 'midfar';
    const frontEdge = DEVIL_BAT_AVG_SPEED * c.t; // how far the leading bats have actually traveled so far
    aliveEnemies().forEach(en=>{
      if(!inRangeBand(en, band)) return;
      const pos = enemyRenderPos(en);
      const distFromOrigin = Math.hypot(pos.x-c.origin.x, pos.y-c.origin.y);
      if(distFromOrigin > frontEdge) return; // the bats simply haven't reached this far yet
      const ang = Math.atan2(pos.y-c.origin.y, pos.x-c.origin.x);
      if(Math.abs(angleDiff(ang, c.coneAngle)) > c.coneHalfAngle) return;
      if(!c.insideMap.has(en)){
        c.insideMap.set(en, {tickT:0});
        c.hitSet.add(en);
        damageEnemy(en, c.dmg, 'special', c.aff);
        en.pausedT = Math.max(en.pausedT||0, c.T.stunDur);
        fx('devilSwarmHit', pos, {});
      } else if(c.T.dotEnabled){
        const rec = c.insideMap.get(en);
        rec.tickT += dt;
        if(rec.tickT >= c.sp.dotInterval){
          rec.tickT -= c.sp.dotInterval;
          damageEnemy(en, c.dmg*(c.sp.dotTick/c.sp.dmgSwarm), 'special', c.aff);
        }
      }
    });
  }
  function resolveDevilClaw(c){
    if(!c.target.alive) return;
    c.comboFired = true;
    const pos = enemyRenderPos(c.target);
    const hitR = 20;
    const clawDmg = c.dmg * (c.sp.dmgClaw/c.sp.dmgSwarm);
    aliveEnemies().forEach(en=>{
      const p2 = enemyRenderPos(en);
      if(Math.hypot(p2.x-pos.x, p2.y-pos.y) <= hitR){
        damageEnemy(en, clawDmg, 'physical', c.aff);
        en.dist = Math.max(0, en.dist - c.sp.clawKnock);
      }
    });
    fx('devilClawHit', pos, {});
  }
  function updateDevilCasts(dt){
    for(let i=state.devilCasts.length-1;i>=0;i--){
      const c = state.devilCasts[i];
      c.t += dt;
      if(c.phase==='manifest'){
        if(c.t >= c.manifestDur){ c.phase='windup'; c.t=0; }
      } else if(c.phase==='windup'){
        if(c.t >= c.windupDur){ c.phase='release'; c.t=0; }
      } else if(c.phase==='release'){
        // continuous stream: keep spawning fresh bats every ~50ms for the entire breathing duration
        c.batSpawnTimer += dt;
        while(c.batSpawnTimer >= 0.05){
          c.batSpawnTimer -= 0.05;
          c.bats.push({spawnT:c.t, angOff:(Math.random()-0.5)*c.coneHalfAngle*1.8,
            speed:170+Math.random()*60, wobble:Math.random()*10, flapOff:Math.random()*10});
        }
        c.bats = c.bats.filter(b => (c.t-b.spawnT)*b.speed < 340); // drop bats once fully faded out
        tickDevilSwarm(c, dt);
        if(c.t >= c.releaseDur){ c.phase='neutral'; c.t=0; }
      } else if(c.phase==='neutral'){
        // bats already in flight keep traveling/fading during this transition, but no new ones spawn
        c.bats = c.bats.filter(b => (c.releaseDur-b.spawnT)*b.speed + b.speed*c.t < 340);
        if(c.comboEligible && !c.comboFired && c.hitSet.has(c.target) && c.t >= c.clawResolveT){
          c.comboFired = true;
          resolveDevilClaw(c);
        }
        if(c.t >= c.neutralDur){ c.phase='fade'; c.t=0; }
      } else if(c.phase==='fade'){
        if(c.t >= c.fadeDur){ state.devilCasts.splice(i,1); }
      }
    }
  }

  function liftOffset(e){
    if(!e.lift) return 0;
    const L = e.lift;
    if(L.phase==='rise') return L.height*easeOutCubic(Math.min(1,L.t/Math.max(0.001,L.rise)));
    if(L.phase==='hold'){
      const bob = L.kind==='bubble' ? Math.sin((L.t/Math.max(0.001,L.hold))*Math.PI*4)*Math.min(3,L.height*0.06) : 0;
      return L.height + bob;
    }
    if(L.phase==='fall') return L.height*(1-easeInCubic(Math.min(1,L.t/Math.max(0.001,L.fall))));
    if(L.phase==='land') return -Math.sin(Math.min(1,L.t/0.22)*Math.PI)*4; // brief squash dip on impact
    return 0;
  }
  function tryJuggle(e){

  function dieColumnLaunchPos(idx){
    const col = idx % 5;
    return {x: cw*(0.14+col*0.18), y: ch*0.97};
  }
  // Approximates "the die's position" as a point at the bottom of the canvas — column mapped
  // across the width, row mapped to a small vertical band right at the bottom edge. This is a
  // deliberate compromise: the board sits in its own DOM region entirely below the canvas, so a
  // canvas context literally cannot draw at the tile's true on-screen position (that coordinate
  // falls outside the canvas's own pixel box and is simply never rendered). Using
  // getBoundingClientRect() here — which is what this used to do — produced a coordinate below
  // the canvas's bottom edge, which is why nothing appeared to spawn from the die at all.
  function dieTileLaunchPos(idx){
    const col = idx % 5, row = Math.floor(idx/5);
    return { x: cw*(0.14+col*0.18), y: ch*(0.90+row*0.033) };

  /* ============================ RANGE BANDS ============================ */
  // A unified 0-1 "how far along has this enemy gotten" reading, regardless of which
  // movement model it uses — normal path-distance, Typhoon's single-lane approach, or a
  // zombie's own top-to-bottom march. Range bands (close/mid/far) are always thirds of THIS.
  function progressFrac(e){
    if(e.isTyphoon || e.isZombie) return e.approach;
    return e.dist/Math.max(1,totalLen);
  }
  function pathFrac(e){ return e.dist/Math.max(1,totalLen); }
  function inRangeBand(e, band){
    if(!band || band==='map') return true;
    const f = progressFrac(e);
    if(band==='close') return f >= 0.65; // nearest third — closest to the tower / dice tray
    if(band==='mid') return f >= 0.30;   // "close, mid" combined — anything past the first third
    if(band==='midfar') return f < 0.65; // "mid, far" combined — everything EXCEPT close range
    return true;
  }
  function filterByRange(list, band){
    if(!band || band==='map') return list;
    const filtered = list.filter(e=>inRangeBand(e,band));
    return filtered.length ? filtered : list; // never leave a die with literally nothing to swing at
  }

  /* ============================ PIERCING SHOTS (Wind Blade, and future line-travel dice) ============================ */
  function spawnPierceShot(originIdx, targetPos, speed, dmg, dmgCategory, radius, cfg){
    const origin = dieTileLaunchPos(originIdx);
    const dx = targetPos.x-origin.x, dy = targetPos.y-origin.y;
    const dist = Math.max(1, Math.hypot(dx,dy));
    const vx = dx/dist*speed, vy = dy/dist*speed;
    const shot = Object.assign({x:origin.x, y:origin.y, vx, vy, dmg, dmgCategory, radius, hitMap:new Map(), t:0, maxT:3}, cfg||{});
    state.pierceShots.push(shot);
    return shot;
  }
  function updatePierceShots(dt){
    for(let i=state.pierceShots.length-1;i>=0;i--){
      const s = state.pierceShots[i];
      s.t += dt;
      if(s.spawnDelay && s.t < s.spawnDelay) continue; // trailing blade hasn't "launched" yet
      s.x += s.vx*dt; s.y += s.vy*dt;
      s.hitMap.forEach((v,k)=>{ const nv=v-dt; if(nv<=0) s.hitMap.delete(k); else s.hitMap.set(k,nv); });
      aliveEnemies().forEach(e=>{
        const p = enemyRenderPos(e);
        const dist = Math.hypot(p.x-s.x, p.y-s.y);
        if(dist <= s.radius && !s.hitMap.has(e)){
          damageEnemy(e, s.dmg, s.dmgCategory, s.affinities);
          s.hitMap.set(e, 0.16);
          if(s.onHit) s.onHit(e, s);
        }
        if(s.vortexRadius && dist <= s.vortexRadius){
          if(s.vortexTickDmg){
            s.vortexTickMap = s.vortexTickMap || new Map();
            const cd = (s.vortexTickMap.get(e)||0) - dt;
            if(cd <= 0){ damageEnemy(e, s.vortexTickDmg, s.dmgCategory, s.affinities); s.vortexTickMap.set(e, 0.35); }
            else s.vortexTickMap.set(e, cd);
          }
          // "drawn in" is represented as a strong, very short slow refreshed each frame while caught —
          // enemies don't have free 2D position, so this is the path-based analog of being pulled toward the vortex.
          applySlow(e, 0.5, 0.2);
        }
      });
      if(s.t>=s.maxT || s.x<-30 || s.x>cw+30 || s.y<-30 || s.y>ch+30) state.pierceShots.splice(i,1);
    }
  }

  /* ============================ CROSSING GATE (path obstacle) ============================ */
  function spawnCrossingGate(target, sp){
    const anchorDist = Math.min(totalLen-2, target.dist + sp.monsterLenGap);
    // find local path tangent at the anchor to decide orientation
    let acc = 0, dx=1, dy=0;
    for(let i=0;i<segLens.length;i++){
      if(anchorDist <= acc+segLens[i] || i===segLens.length-1){
        const a=pathPts[i], b=pathPts[i+1];
        dx = b.x-a.x; dy = b.y-a.y; break;
      }
      acc += segLens[i];
    }
    const vertical = Math.abs(dy) >= Math.abs(dx);
    // Pick which side the post sits on based on the ACTUAL direction of travel at this point
    // (enemies always move from dist 0 -> totalLen, so this sign is the real march direction),
    // so the arm always swings out to meet oncoming enemies correctly no matter which way
    // this stretch of path happens to run.
    const pivotSide = vertical ? (dy>=0 ? 'right' : 'left') : (dx>=0 ? 'top' : 'bottom');
    const pos = posAtDistance(anchorDist);
    state.gate = {
      dist: anchorDist, x:pos.x, y:pos.y, vertical, pivotSide,
      hp: sp.gateHp, maxHp: sp.gateHp, dmgTakenMult: sp.dmgTakenMult,
      phase:'lowering', t:0, lowerDur: sp.lowerDur, bigBossDmg: sp.bigBossDmg,
      hitFlash:0, atkTickT:0,
    };
  }
  function gateTakeDamage(amount){
    if(!state.gate) return;
    state.gate.hp -= amount * state.gate.dmgTakenMult;
    state.gate.hitFlash = 0.15;
    if(state.gate.hp <= 0){
      fx('explosion', {x:state.gate.x, y:state.gate.y});
      state.gate = null;
      state.gateCooldownT = 3.12;
    }
  }
  function updateGate(dt){
    if(state.gateCooldownT>0) state.gateCooldownT -= dt;
    if(!state.gate) return;
    const g = state.gate;
    if(g.hitFlash>0) g.hitFlash -= dt;
    if(g.phase==='lowering'){
      g.t += dt;
      if(g.t >= g.lowerDur){ g.phase='active'; }
      return;
    }
    // active: block normal/small enemies at the gate line; sub/typhoon smash through for low damage
    g.atkTickT += dt;
    const tickNow = g.atkTickT >= 1.0;
    if(tickNow) g.atkTickT = 0;
    for(const e of state.enemies){
      if(!e.alive) continue;
      const bigBoss = e.tier==='sub' || e.isTyphoon;
      if(e.dist >= g.dist - 1){
        if(bigBoss){
          if(!e._gateSmashed){ e._gateSmashed = true; gateTakeDamage(g.bigBossDmg); }
        } else {
          e.dist = Math.min(e.dist, g.dist - 1);
          if(tickNow && state.gate){ gateTakeDamage(e.atk||3); e.hitFlash = 0.12; }
        }
      }
    }
  }

  function scaledHp(kind, waveNum){

  function cumulativeRoster(stage, wave){
    let pool = (stage.carryover||[]).slice();
    Object.keys(stage.introduce).map(Number).filter(w=>w<=wave).sort((a,b)=>a-b).forEach(w=>{ pool.push(...stage.introduce[w]); });
    return [...new Set(pool)];
  }
  function buildAdventureWave(stage, wave, diff){
    const pool = cumulativeRoster(stage, wave);
    const baseCount = 4 + Math.floor(wave*0.6) + diff.extraMonsters;
    const q = [];
    for(let i=0;i<baseCount;i++) q.push({key: pool[Math.floor(Math.random()*pool.length)], tier:'normal', delay:0.5});
    if(stage.smallBoss && stage.smallBoss[wave]) q.push({key: stage.smallBoss[wave], tier:'small', delay:1.2});
    if(stage.subBoss && stage.subBoss.wave === wave) q.push({key: stage.subBoss.key, tier:'sub', delay:1.5});
    return q;
  }

  // Slow, noticeable ramp keyed off BOTH elapsed playtime and kill count together, capped so it
  // never spirals indefinitely. ~10 minutes of play or ~150 kills (whichever the player reaches
  // first, and they compound) roughly doubles zombie toughness and spawn pressure.
  function zombieGrowthFactor(){
    return Math.min(2.2, (state.zPlayTime/600) + (state.kills/150));
  }
  // Each stat unlocks at its own kill-count gate, then ramps in linearly over the following
  // `window` kills (rather than snapping on instantly) before compounding with the general
  // growth factor above. Staggered on purpose: HP first (~20 kills), then damage resistance
  // (~50), then status/effect resistance (~90) — offensive power rides the same early gate as HP.
  function killRamp(kills, threshold, window){
    if(kills < threshold) return 0;
    return Math.min(1, (kills-threshold)/window);
  }
  function computeZombieScale(){
    const g = zombieGrowthFactor();
    const k = state.kills;
    const hpRamp = killRamp(k, 20, 70);
    const dmgRamp = killRamp(k, 20, 70);
    const spdRamp = killRamp(k, 20, 70);
    const resistRamp = killRamp(k, 50, 90);
    const statusRamp = killRamp(k, 90, 130);
    return {
      hpMult: 1 + g*0.65*hpRamp,
      dmgMult: 1 + g*0.40*dmgRamp,
      speedMult: 1 + Math.min(g,1.2)*0.15*spdRamp,
      dmgResist: Math.min(0.65, g*0.35*resistRamp),   // hard cap — damage reduction never exceeds 65%
      statusResist: Math.min(1.0, g*0.55*statusRamp), // can reach full 100% immunity eventually
    };
  }
  // A player-facing "Level" — deliberately decoupled from the combat-stat curve above, which
  // saturates by design. Level 1 covers kills 0-19 (before any stat has moved at all); Level 2
  // lands exactly at kill 20, matching HP's onset; from there it climbs steadily off BOTH kills
  // and playtime so a long, dedicated run keeps seeing the number go up even once the underlying
  // multipliers have already capped out. Purely a badge/reference — never re-evaluated after
  // a zombie spawns, same as its baked-in stats.
  function zombieLevelForSpawn(){
    const k = state.kills;
    if(k < 20) return 1;
    const progress = (k-20) + state.zPlayTime/15;
    return Math.min(999, 2 + Math.floor(progress/6));
  }
  function availableZombieTypes(){
    const pool = ['wanderer_guy','wanderer_gal','wanderer_gob'];

    return pool;
  }
  function zombieSpawnInterval(){ return Math.max(0.9, 3.2 - zombieGrowthFactor()*1.1); }
  function zombiesPerSpawnBurst(){ return 1 + Math.floor(zombieGrowthFactor()*1.8); }
  function updateZombieSpawning(dt){
    state.zombieScale = computeZombieScale();
    state.zSpawnTimer += dt;
    const interval = zombieSpawnInterval();
    if(state.zSpawnTimer >= interval){
      state.zSpawnTimer -= interval;
      const pool = availableZombieTypes();
      const n = zombiesPerSpawnBurst();
      for(let i=0;i<n;i++) spawnZombie(pool[Math.floor(Math.random()*pool.length)]);
    }
  }

  }
  function updateZombieMovement(z, dt){
    if(z.walkingToLane!=null){
      // actively changing lanes: real horizontal walk at normal speed, no forward progress meanwhile
      const targetX = zombieColumnX(z.walkingToLane);
      const dir = targetX>z.x ? 1 : -1;
      z.x += dir*z.speed*dt;
      if((dir>0 && z.x>=targetX) || (dir<0 && z.x<=targetX)){
        z.x = targetX; z.lane = z.walkingToLane; z.walkingToLane = null;
      }
      return;
    }
    const engageBand = z.def.engageBand || 'close';
    const engageAt = engageBand==='close' ? 0.90 : 0.55; // policeman stops in the mid band, not right at the tray
    if(z.approach < engageAt){
      z.approach = Math.min(1, z.approach + (z.speed/260)*dt);
      z.engaged = false;
    } else {
      z.engaged = true;
    }
  }
  function zombieOutgoingDmg(z, s){ return (s.dmg||0) * (z.dmgMult||1); }

    const targetIdx = z.targetIdx;
    if(targetIdx==null || !state.board[targetIdx]) return;
    const kind = ZOMBIE_FX_KIND[s.type];
    if(!kind) return;

  function resolveZombieAttack(fx){
    const z = fx.z, s = fx.s, targetIdx = fx.targetIdx;
    if(!state.board[targetIdx]) return; // die moved/was destroyed mid-animation — attack simply whiffs
    atkCategory = s.category || 'physical'; atkElement = s.element || null;
    const dmg = zombieOutgoingDmg(z, s);
    switch(fx.kind){
      case 'clawSwipe': {
        dieDamage(targetIdx, dmg, 'pulse-claw');
        break;
      }
      case 'doubleClaw': {
        dieDamage(targetIdx, dmg, 'pulse-claw');
        const d = state.board[targetIdx];
        if(d) d.disabledT = Math.max(d.disabledT||0, s.stunTarget||0);
        break;
      }
      case 'helmetJab': {
        dieDamage(targetIdx, dmg, 'pulse-hit');
        break;
      }
      case 'acidVomit': {
        dieDamage(targetIdx, dmg, 'pulse-pollen');
        const d = state.board[targetIdx];
        if(d && Math.random()<s.poisonChance && !d.poisonZ){
          d.poisonZ = {t:s.poisonDur, tickT:0, perTick:s.poisonPerTick};
        }
        state.acidGlobs.push({idx:targetIdx, life:s.globLife, tickT:0, chance:s.globPoisonChance, interval:s.globInterval, perTick:s.poisonPerTick, dur:s.poisonDur});
        break;
      }
      case 'gunShot': {
        dieDamage(targetIdx, dmg, 'pulse-hit');
        break;
      }
    }
  }
